\entry{Pennsylvania State University}{Sep.\ 2024 -- Present}{Graduate Researcher, Agentic AI \& Multi-Agent LLM Systems}{University Park, PA}
\begin{cvlist}
  \proj{InsightProv}{Qwen2.5-72B, LoRA, adversarial training}{Insight-provenance detection for scientific reviews: identifies whether a review's insights come from humans, LLMs, or both. Built \textbf{InsightProv-v0} (12,000+ reviews on 4,000+ papers, sentence-level labels); a two-stage adversarial framework unlearns stylistic authorship shortcuts. Found that AI insights stay close to generic or paper-provided content, while human insights add external knowledge and independent judgment.}
  \proj{DIAGPaper}{Python, GPT-4o, Claude, Llama, LangChain}{Multi-agent framework for automated peer review. A Customizer generates paper-specific criteria; a Rebuttal module runs author--reviewer debate, filtering 40--60\% of invalid critiques; a Prioritizer, learned from 15,343 real reviews, ranks weaknesses by severity. \textbf{F1 51.89 on AAAR, 9.4\% above GPT-4o.}}
  \proj{AAAR-1.0}{Python, benchmark design}{Large-scale benchmark for AI-assisted research. \textbf{ICML 2025 Workshop Best Paper Award.}}
  \proj{M2-Verify}{VLMs, multimodal reasoning}{Multidomain benchmark for claim consistency across text and images.}
  \proj{Know-Act Gap}{LLM agents, autoregressive reasoning}{Task-level autoregressive reasoning that turns LLM knowledge into executable actions in agentic settings.}
\end{cvlist}
